## 1. Project Overview

**Kaagapay** is a free, mobile-first web application that helps residents of Silang, Cavite and the broader CALABARZON region (Region 4A) find verified social services, government programs, health centers, legal aid offices, and community resources.

### The Problem
People in need have no single, reliable, up-to-date directory of free services available to them. Information is scattered across LGU websites, Facebook pages, and word of mouth.

### The Solution
A searchable, map-enabled, community-maintained directory of social services. Anyone can search. Community members can submit new resources. An admin reviews and publishes submissions.

### Language Strategy
- **Default language: English** — broader reach, portfolio-friendly
- **Toggle button in navbar** — switches all UI text to Filipino/Tagalog instantly
- Language preference saved to `localStorage` — persists across sessions
- No page reload needed — handled via React context

### Target Users
- Residents of Silang, Cavite (primary)
- Residents of broader Region 4A (secondary)
- Barangay health workers and social workers (power users)

## 6. Drizzle Schema

**File: `backend/src/db/schema.ts`**

```typescript
import {
  pgTable,
  pgEnum,
  uuid,
  varchar,
  text,
  boolean,
  numeric,
  timestamp,
  index,
} from "drizzle-orm/pg-core";
import { sql } from "drizzle-orm";

// ─── Enums ────────────────────────────────────────────────────────────────────

export const categoryEnum = pgEnum("category_type", [
  "kalusugan",     // Health
  "legal_aid",     // Legal Aid
  "pagkain",       // Food & Relief
  "kabuhayan",     // Livelihood
  "mental_health", // Mental Health
  "edukasyon",     // Education
  "pabahay",       // Housing
  "pamilya",       // Children & Family
]);

export const costEnum = pgEnum("cost_type", [
  "libre",     // Free
  "may_bayad", // With fee
  "depende",   // Depends
]);

export const statusEnum = pgEnum("status_type", [
  "active",
  "inactive",
  "pending",
  "rejected",
]);

// ─── Resources ────────────────────────────────────────────────────────────────

export const resources = pgTable(
  "resources",
  {
    id: uuid("id")
      .primaryKey()
      .default(sql`gen_random_uuid()`),

    name:         varchar("name", { length: 255 }).notNull(),
    organization: varchar("organization", { length: 255 }),

    category: categoryEnum("category").notNull(),

    address:  text("address").notNull(),
    barangay: varchar("barangay", { length: 100 }),
    city:     varchar("city", { length: 100 }).notNull().default("Silang"),
    province: varchar("province", { length: 100 }).notNull().default("Cavite"),

    phone:   varchar("phone", { length: 50 }),
    hours:   varchar("hours", { length: 255 }),
    website: varchar("website", { length: 255 }),

    cost:        costEnum("cost").notNull().default("libre"),
    description: text("description"),

    // Coordinates for Leaflet map pins
    lat: numeric("lat", { precision: 10, scale: 7 }),
    lng: numeric("lng", { precision: 10, scale: 7 }),

    status:   statusEnum("status").notNull().default("active"),
    verified: boolean("verified").notNull().default(false),

    createdAt: timestamp("created_at", { withTimezone: true })
      .notNull()
      .default(sql`now()`),
    updatedAt: timestamp("updated_at", { withTimezone: true })
      .notNull()
      .default(sql`now()`),
  },
  (table) => ({
    categoryIdx: index("idx_resources_category").on(table.category),
    statusIdx:   index("idx_resources_status").on(table.status),
    cityIdx:     index("idx_resources_city").on(table.city),
  })
);

// ─── Submissions ──────────────────────────────────────────────────────────────

export const submissions = pgTable(
  "submissions",
  {
    id: uuid("id")
      .primaryKey()
      .default(sql`gen_random_uuid()`),

    // Resource info — mirrors resources table (no lat/lng until admin approves)
    name:         varchar("name", { length: 255 }).notNull(),
    organization: varchar("organization", { length: 255 }),
    category:     categoryEnum("category").notNull(),
    address:      text("address").notNull(),
    barangay:     varchar("barangay", { length: 100 }),
    phone:        varchar("phone", { length: 50 }),
    hours:        varchar("hours", { length: 255 }),
    cost:         costEnum("cost").notNull().default("libre"),
    description:  text("description"),

    // Who submitted it (optional — anonymous allowed)
    submitterName:  varchar("submitter_name", { length: 255 }),
    submitterEmail: varchar("submitter_email", { length: 255 }),

    // Review fields
    status:     statusEnum("status").notNull().default("pending"),
    reviewedAt: timestamp("reviewed_at", { withTimezone: true }),

    createdAt: timestamp("created_at", { withTimezone: true })
      .notNull()
      .default(sql`now()`),
  },
  (table) => ({
    statusIdx: index("idx_submissions_status").on(table.status),
  })
);

// ─── Admins ───────────────────────────────────────────────────────────────────

export const admins = pgTable("admins", {
  id: uuid("id")
    .primaryKey()
    .default(sql`gen_random_uuid()`),

  username:     varchar("username", { length: 100 }).notNull().unique(),
  passwordHash: text("password_hash").notNull(),

  createdAt: timestamp("created_at", { withTimezone: true })
    .notNull()
    .default(sql`now()`),
});

// ─── Inferred Types (use these everywhere instead of raw SQL types) ────────────

export type Resource      = typeof resources.$inferSelect;
export type NewResource   = typeof resources.$inferInsert;
export type Submission    = typeof submissions.$inferSelect;
export type NewSubmission = typeof submissions.$inferInsert;
export type Admin         = typeof admins.$inferSelect;
export type NewAdmin      = typeof admins.$inferInsert;

export type Category = typeof categoryEnum.enumValues[number];
export type Cost     = typeof costEnum.enumValues[number];
export type Status   = typeof statusEnum.enumValues[number];
```

### Public Endpoints

|--------|-------|-------------|
| `GET` | `/api/resources` | List resources. Query: `category`, `city`, `barangay`, `cost`, `q`, `lat`, `lng` |
| `GET` | `/api/resources/:id` | Single resource by ID |
| `GET` | `/api/categories` | All categories with resource counts |
| `POST` | `/api/submissions` | Submit a new resource for review |

### Admin Endpoints (JWT required)

| Method | Route | Description |
|--------|-------|-------------|
| `POST` | `/api/admin/login` | Returns JWT |
| `GET` | `/api/admin/resources` | All resources including inactive |
| `POST` | `/api/admin/resources` | Create resource directly |
| `PUT` | `/api/admin/resources/:id` | Update resource |
| `PATCH` | `/api/admin/resources/:id/status` | Toggle active/inactive |
| `GET` | `/api/admin/submissions` | List pending submissions |
| `PATCH` | `/api/admin/submissions/:id/approve` | Approve → auto-creates resource |
| `PATCH` | `/api/admin/submissions/:id/reject` | Reject and archive |
| `GET` | `/api/admin/stats` | Dashboard counts |

### Example Query with Drizzle

```typescript
// backend/src/db/queries/resources.ts
import { db } from "../client";
import { resources } from "../schema";
import { eq, and, ilike, sql } from "drizzle-orm";

export async function getResources(filters: {
  category?: string;
  city?: string;
  q?: string;
  cost?: string;
}) {
  return db
    .select()
    .from(resources)
    .where(
      and(
        eq(resources.status, "active"),
        filters.category
          ? eq(resources.category, filters.category as any)
          : undefined,
        filters.city
          ? eq(resources.city, filters.city)
          : undefined,
        filters.cost
          ? eq(resources.cost, filters.cost as any)
          : undefined,
        filters.q
          ? ilike(resources.name, `%${filters.q}%`)
          : undefined
      )
    )
    .orderBy(resources.createdAt);
}
```

### Language Toggle Button (in `Navbar.tsx`)

```tsx
import { useTranslation } from "react-i18next";

export function LanguageToggle() {
  const { i18n } = useTranslation();

  const toggle = () => {
    const next = i18n.language === "en" ? "tl" : "en";
    i18n.changeLanguage(next);
    localStorage.setItem("kaagapay-lang", next);
  };

  return (
    <button
      onClick={toggle}
      className="flex items-center gap-1 px-3 py-1.5 rounded-lg
                 border border-white/20 text-white/80 text-sm font-semibold
                 hover:bg-white/10 transition-colors"
    >
      {i18n.language === "en" ? "🇵🇭 Filipino" : "🇺🇸 English"}
    </button>
  );
}
```

## 11. Map Integration

### Library
**Leaflet.js + OpenStreetMap** — free, no API key, no billing.

### `MapView.tsx`

```typescript
import L from "leaflet";
import "leaflet/dist/leaflet.css";
import { useEffect, useRef } from "react";
import type { Resource } from "../../../backend/src/db/schema";

const SILANG_CENTER: [number, number] = [14.2273, 120.9741];

const CATEGORY_COLORS: Record<string, string> = {
  kalusugan:     "#0B2D6E",
  legal_aid:     "#8B5A00",
  pagkain:       "#1A6B3C",
  kabuhayan:     "#5B2D9E",
  mental_health: "#CE1127",
  edukasyon:     "#185FA5",
  pabahay:       "#B45309",
  pamilya:       "#0369A1",
};

interface MapViewProps {
  resources: Resource[];
  userLocation?: [number, number];
}

export function MapView({ resources, userLocation }: MapViewProps) {
  const mapRef      = useRef<HTMLDivElement>(null);
  const mapInstance = useRef<L.Map | null>(null);

  // Init map once
  useEffect(() => {
    if (!mapRef.current || mapInstance.current) return;

    const map = L.map(mapRef.current).setView(SILANG_CENTER, 14);

    L.tileLayer("https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png", {
      attribution: '© <a href="https://openstreetmap.org">OpenStreetMap</a> contributors',
      maxZoom: 19,
    }).addTo(map);

    mapInstance.current = map;
    return () => { map.remove(); mapInstance.current = null; };
  }, []);

  // Add pins when resources change
  useEffect(() => {
    const map = mapInstance.current;
    if (!map) return;

    resources.forEach((r) => {
      if (!r.lat || !r.lng) return;

      L.circleMarker([Number(r.lat), Number(r.lng)], {
        radius:      10,
        fillColor:   CATEGORY_COLORS[r.category] ?? "#0B2D6E",
        color:       "#ffffff",
        weight:      2,
        fillOpacity: 1,
      })
        .addTo(map)
        .bindPopup(`
          <div style="min-width:170px;font-family:sans-serif">
            <strong style="font-size:13px">${r.name}</strong><br/>
            <span style="font-size:11px;color:#6b7280">${r.barangay ?? ""}, ${r.city}</span><br/>
            <span style="font-size:11px;color:#6b7280">${r.hours ?? ""}</span>
            <div style="display:flex;gap:6px;margin-top:8px">
              <a href="tel:${r.phone}"
                 style="flex:1;text-align:center;background:#0B2D6E;color:#fff;
                        padding:5px;border-radius:6px;font-size:12px;text-decoration:none">
                📞 Call
              </a>
              <a href="https://maps.google.com/?q=${r.lat},${r.lng}" target="_blank"
                 style="flex:1;text-align:center;background:#F7A800;color:#0B2D6E;
                        padding:5px;border-radius:6px;font-size:12px;text-decoration:none">
                🗺 Directions
              </a>
            </div>
          </div>
        `);
    });
  }, [resources]);

  // User location dot
  useEffect(() => {
    const map = mapInstance.current;
    if (!map || !userLocation) return;

    L.circleMarker(userLocation, {
      radius: 8, fillColor: "#4285F4",
      color: "#fff", weight: 2.5, fillOpacity: 1,
    }).addTo(map);

    map.setView(userLocation, 15);
  }, [userLocation]);

  return <div ref={mapRef} style={{ height: "100%", width: "100%" }} />;
}
```

> **Critical:** `import "leaflet/dist/leaflet.css"` must be present or map tiles render broken.

## 13. Authentication

Admin-only. Public users have no accounts.

```typescript
// backend/src/middleware/auth.ts
import { Request, Response, NextFunction } from "express";
import jwt from "jsonwebtoken";

export const requireAuth = (req: Request, res: Response, next: NextFunction) => {
  const token = req.headers.authorization?.split(" ")[1];
  if (!token) return res.status(401).json({ success: false, error: "Unauthorized" });

  try {
    const payload = jwt.verify(token, process.env.JWT_SECRET!);
    (req as any).admin = payload;
    next();
  } catch {
    return res.status(401).json({ success: false, error: "Invalid or expired token" });
  }
};
```

**Create first admin (run once):**

```bash
# In your backend directory, run this to get a hash:
node -e "const b=require('bcryptjs'); console.log(b.hashSync('your_password_here', 10))"
```

Then insert via Drizzle seed or Neon SQL editor:
```sql
INSERT INTO admins (username, password_hash)
VALUES ('admin', '<paste_hash_here>');
```
